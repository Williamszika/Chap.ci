# =============================================================================
#  Chap.ci — En-têtes de sécurité pour la RACINE du site
#  À placer/fusionner dans  public_html/.htaccess  (crée le fichier s'il
#  n'existe pas). Les endpoints /api/ ont déjà leurs propres en-têtes ; ceci
#  couvre les pages du site (le frontend).
#
#  ⚠️ Si tu as DÉJÀ un public_html/.htaccess avec des règles, n'écrase pas le
#  fichier : ajoute juste le bloc ci-dessous à la fin.
# =============================================================================
<IfModule mod_headers.c>
  # Empêche le navigateur de « deviner » un type de fichier (anti-sniffing).
  Header always set X-Content-Type-Options "nosniff"
  # Empêche l'affichage du site dans une iframe d'un autre domaine (anti-clickjacking).
  Header always set X-Frame-Options "SAMEORIGIN"
  # Ne divulgue pas l'URL complète en referrer vers les autres sites.
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  # Restreint la géolocalisation au site lui-même ; le reste garde les défauts.
  Header always set Permissions-Policy "geolocation=(self)"
  # Force HTTPS pendant 6 mois (le site est déjà en HTTPS).
  Header always set Strict-Transport-Security "max-age=15552000"
</IfModule>

# --- Cache des fichiers versionnés (perf : 2e visite = zéro retéléchargement) ---
# Vite renomme chaque fichier avec un hash à chaque build → un cache très long est
# sûr (un changement de contenu = un nouveau nom, jamais servi périmé).
<IfModule mod_headers.c>
  <FilesMatch "\.(js|mjs|css|woff2?|ttf|otf|eot)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  # EXCEPTIONS : le service worker et les scripts PWA ne sont PAS hashés et doivent
  # pouvoir se mettre à jour → jamais de cache long, sinon l'app ne se met plus à jour.
  <FilesMatch "^(sw|registerSW|workbox-.+)\.js$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
</IfModule>
